cmake_minimum_required(VERSION 3.20)
project(proton_needle_helper LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build this project with an x86_64 Windows/MinGW toolchain")
endif()

option(BUILD_TESTING "Build fixture-only helper and Windows test fixtures" OFF)
include(cmake/json_headers.cmake)
file(GLOB_RECURSE JSON_HEADERS CONFIGURE_DEPENDS "${NLOHMANN_INCLUDE_DIR}/nlohmann/*.hpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${JSON_HEADERS})
file(COPY "${NLOHMANN_INCLUDE_DIR}/nlohmann"
    DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/vendor_include")
add_executable(proton_needle_helper src/main.cpp)
set_target_properties(proton_needle_helper PROPERTIES OUTPUT_NAME proton_needle_helper_x64)
target_compile_features(proton_needle_helper PRIVATE cxx_std_20)
target_include_directories(proton_needle_helper PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/vendor_include")
target_compile_definitions(proton_needle_helper PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
if(MINGW)
    target_compile_options(proton_needle_helper PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    target_link_options(proton_needle_helper PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(BUILD_TESTING)
    add_executable(proton_needle_fixture_helper src/main.cpp)
    set_target_properties(proton_needle_fixture_helper PROPERTIES OUTPUT_NAME proton_needle_fixture_helper_x64)
    target_compile_features(proton_needle_fixture_helper PRIVATE cxx_std_20)
    target_include_directories(proton_needle_fixture_helper PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/vendor_include")
    target_compile_definitions(proton_needle_fixture_helper PRIVATE
        PROTON_NEEDLE_FIXTURE_TESTING=1 WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
    if(MINGW)
        target_compile_options(proton_needle_fixture_helper PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        target_link_options(proton_needle_fixture_helper PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../fixtures" "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
endif()
